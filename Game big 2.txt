function TB(x0,y0){
  const pts=[[x0,y0]]; let cx=x0, cy=y0;
  const api={
    flat(len){ cx+=len; pts.push([cx,cy]); return api; },
    slope(len,dy){ const n=Math.max(2,Math.ceil(Math.abs(len)/3));
      for(let i=1;i<=n;i++) pts.push([cx+len*i/n, cy+dy*i/n]);
      cx+=len; cy+=dy; return api; },
    bump(len,h){ const n=9; for(let i=1;i<=n;i++){const tt=i/n; pts.push([cx+len*tt, cy+Math.sin(tt*Math.PI)*h]);} cx+=len; pts.push([cx,cy]); return api; },
    dip(len,d){ return api.bump(len,-d); },
    gap(len,depth){ depth=depth||30; pts.push([cx+0.06,cy-depth]); pts.push([cx+len-0.06,cy-depth]); cx+=len; pts.push([cx,cy]); return api; },
    steps(n,w,h){ for(let i=0;i<n;i++){cx+=w;cy+=h;pts.push([cx,cy]);} return api; },
    get x(){return cx;}, get y(){return cy;},
    done(){ return pts; }
  };
  return api;
}

function scatterObstacles(rng,pts,xStart,xEnd,types,count,diff){
  const out=[];
  for(let i=0;i<count;i++){
    const x=lerp(xStart,xEnd,(i+0.5+ (rng()-0.5)*0.6)/count);
    if(x<pts[0][0]+18) continue;
    const type=types[Math.floor(rng()*types.length)];
    const o={t:type,x:x,ground:true};
    if(type==='box'){ o.w=1.1+rng()*0.9; o.h=0.9+rng()*1.2; }

function mat(color,opts={}){ return new THREE.MeshStandardMaterial(Object.assign({color:color,roughness:0.7,metalness:0.1},opts)); }

function buildGroundMesh(pts,theme,W){
  const T=THEMES[theme];
  const g=new THREE.Group();
  const topMat=mat(T.ground,{roughness:0.92,metalness:0.05});
  const sideMat=mat(T.ground2,{roughness:0.95});
  const positions=[],normalsArr=[],colors=[];
  const verts=[];
  const push=(a,b,c,d)=>{ // quad a,b,c,d (ccw)
    verts.push(a.x,a.y,a.z,b.x,b.y,b.z,c.x,c.y,c.z);
    verts.push(a.x,a.y,a.z,c.x,c.y,c.z,d.x,d.y,d.z);
  };
  const V=(x,y,z)=>new THREE.Vector3(x,y,z);
  for(let i=0;i<pts.length-1;i++){
    const A=pts[i],B=pts[i+1];
    const len=Math.hypot(B[0]-A[0],B[1]-A[1]);
    if(len<0.001) continue;
    // top surface quad
    push(V(A[0],A[1],-W),V(A[0],A[1],W),V(B[0],B[1],W),V(B[0],B[1],-W));
    // side skirts
    const depth=3.0;
    push(V(A[0],A[1],-W),V(B[0],B[1],-W),V(B[0],B[1]-depth,-W),V(A[0],A[1]-depth,-W));
    push(V(A[0],A[1],W),V(A[0],A[1]-depth,W),V(B[0],B[1]-depth,W),V(B[0],B[1],W));
  }
  const geo=new THREE.BufferGeometry();
  geo.setAttribute('position',new THREE.Float32BufferAttribute(verts,3));
  geo.computeVertexNormals();
  const m=new THREE.Mesh(geo,topMat);
  m.receiveShadow=true; m.castShadow=false;
  m.userData.shared=false;
  g.add(m);
  return g;
}

    const frame=new THREE.Mesh(new THREE.BoxGeometry(0.5,h,w),mat(0x556070,{metalness:0.7,roughness:0.4}));
    frame.castShadow=true;
    const blade=new THREE.Group();
    for(let i=0;i<4;i++){
      const b=new THREE.Mesh(new THREE.BoxGeometry(0.12,h*0.42,w*0.16),mat(0x8899aa,{metalness:0.6}));
      b.rotation.x=i*Math.PI/2;
      b.position.set(0,Math.cos(i*Math.PI/2)*h*0.2, Math.sin(i*Math.PI/2)*w*0.2);
      blade.add(b);
    }
    blade.rotation.y=Math.PI/2;
    blade.position.x=0.3;
    g.add(frame); g.add(blade);
    g.userData.blade=blade;
  } else if(o.t==='boost'){
    const m=new THREE.Mesh(new THREE.BoxGeometry(o.w||2,0.12,2.2),mat(0x22ccff,{emissive:0x00aaff,emissiveIntensity:1.5,transparent:true,opacity:0.85}));
    g.add(m);
  }
  g.traverse(x=>{ if(x.isMesh){ x.castShadow=(quality!=='low'); x.receiveShadow=true; } });
  return g;
}

/* IK solver: given hip pivot (parent space) and target (parent space), solve */
function solveIK(limb,tx,ty,l1,l2,flip){
  let dx=tx, dy=ty;
  let d=Math.hypot(dx,dy);
  d=clamp(d,Math.abs(l1-l2)+0.02,l1+l2-0.01);
  const alpha=Math.atan2(dy,dx);
  const cosB=(d*d+l1*l1-l2*l2)/(2*d*l1);
  const B=Math.acos(clamp(cosB,-1,1));
  const phiThigh=alpha+B*(flip?-1:1);
  const cosG=(l1*l1+l2*l2-d*d)/(2*l1*l2);
  const G=Math.acos(clamp(cosG,-1,1));
  const phiShin=phiThigh-(Math.PI-G)*(flip?-1:1);
  limb.hipPivot.rotation.z=phiThigh+Math.PI/2;
  limb.kneePivot.rotation.z=phiShin-phiThigh;
}

const P={

const bike={

  onGround:false, groundCount:0,
  wheelSpin:0, crank:0, airTime:0, boost:0,
  crashTimer:0, landed:false

let level=null;
let platformStates=[];  // dynamic state for movers etc
let activeObstacles=[];

function buildWorldState(lv){
  activeObstacles=lv.obstacles||[];
  platformStates=activeObstacles.map((o,i)=>{
    const st={o,i,time:Math.random()*6, y:0, x:0, vy:0, vx:0};
    if(o.t==='mover'){
      if(o.ground){
        st.baseY=(groundAt(lv,o.x)||0)+(o.h/2)+0.2;
      } else st.baseY=o.py!==undefined?o.py:(o.y||-2);
      st.x=o.x; st.y=st.baseY;
      st.span=o.dy||3;
    }
    if(o.t==='saw'){ st.angle=0; }
    if(o.t==='hammer'){ st.angle=-0.6; }
    if(o.t==='spring'){ st.cool=0; st.squash=0; }
    return st;
  });
}

function terrainY(pts,x){
  let best=-Infinity;
  for(let i=0;i<pts.length-1;i++){
    const a=pts[i],b=pts[i+1];
    if(x>=a[0]&&x<=b[0]){
      const tt=(x-a[0])/((b[0]-a[0])||1);
      const y=a[1]+(b[1]-a[1])*tt;
      if(y>best) best=y;
    }
  }
  return best;
}
function groundAt(lv,x){ return terrainY(lv.pts,x); }

function terrainNormal(pts,x){
  for(let i=0;i<pts.length-1;i++){
    const a=pts[i],b=pts[i+1];
    if(x>=a[0]&&x<=b[0]){
      const dx=b[0]-a[0],dy=b[1]-a[1];
      const L=Math.hypot(dx,dy)||1;
      return {x:-dy/L,y:dx/L};
    }
  }
  return {x:0,y:1};
}

/* find highest surface below (x, yMax) */
function surfaceBelow(x,yMax){
  let bestY=-Infinity, norm={x:0,y:1};
  const pts=level.pts;
  for(let i=0;i<pts.length-1;i++){
    const a=pts[i],b=pts[i+1];
    if(x>=a[0]&&x<=b[0]){
      const tt=(x-a[0])/((b[0]-a[0])||1);
      const y=a[1]+(b[1]-a[1])*tt;
      if(y<=yMax+0.35 && y>bestY){
        bestY=y;
        const dx=b[0]-a[0],dy=b[1]-a[1],L=Math.hypot(dx,dy)||1;
        norm={x:-dy/L,y:dx/L};
      }
    }
  }
  // moving platforms (rideable)
  for(const st of platformStates){
    const o=st.o;
    if(o.t!=='mover') continue;
    const top=st.y+(o.h||0.7)/2;
    if(x>=st.x-(o.w||3.5)/2 && x<=st.x+(o.w||3.5)/2){
      if(top<=yMax+0.35 && top>bestY){ bestY=top; norm={x:0,y:1}; }
    }
  }
  return {y:bestY,norm};
}

function frictionAt(x){
  if(level.zones){ for(const z of level.zones){ if(x>=z.x0&&x<=z.x1) return z.mu; } }
  return 1.0;
}

function resetBikeAt(x,y){
  bike.x=x; bike.y=y; bike.vx=0; bike.vy=0; bike.ang=0; bike.angVel=0;
  bike.wheelSpin=0; bike.airTime=0; bike.crashTimer=0; bike.boost=0;
}

/* fixed-step physics */
const FIXED=1/120;
function stepPhysics(dt){
  const m=P.mass, I=P.I;
  let fx=0,fy=0,torque=0;

  fy-=m*P.gravity;

  // air drag
  const sp=Math.hypot(bike.vx,bike.vy);
  fx-=bike.vx*Math.abs(bike.vx)*P.drag;
  fy-=bike.vy*Math.abs(bike.vy)*P.drag*0.4;

  const cosA=Math.cos(bike.ang), sinA=Math.sin(bike.ang);
  const wheelDefs=[
    {lx:-P.wheelBase/2, ly:P.wheelY, drive:true},
    {lx: P.wheelBase/2, ly:P.wheelY, drive:false}
  ];
  let anyGround=false;

  const input=Game.input;
  const throttle=input.gas*SaveManager.data.settings.ctlSens;
  const braking=input.brake;
  const lean=input.lean; // -1 back, +1 forward

  for(const w of wheelDefs){
    const wx=bike.x + w.lx*cosA - w.ly*sinA;
    const wy=bike.y + w.lx*sinA + w.ly*cosA;
    const surf=surfaceBelow(wx, wy+P.wheelR*0.2);
    if(surf.y===-Infinity) continue;
    const wheelBottom=wy-P.wheelR;
    const comp=surf.y-wheelBottom;
    if(comp<=0) continue;

    anyGround=true;
    const n=surf.norm;
    // velocity at wheel point
    const rx=wx-bike.x, ry=wy-bike.y;
    const vwx=bike.vx - bike.angVel*ry;
    const vwy=bike.vy + bike.angVel*rx;
    const vn=vwx*n.x+vwy*n.y;

    let Fn=P.suspK*Math.min(comp,0.7) - P.suspD*vn;
    Fn=Math.max(0,Fn);

    fx+=Fn*n.x; fy+=Fn*n.y;
    torque+=rx*(Fn*n.y)-ry*(Fn*n.x);

    // tangential
    const tx=n.y, ty=-n.x;
    const vt=vwx*tx+vwy*ty;
    const mu=frictionAt(wx);
    let Ft=0;

    if(w.drive && throttle>0){
      const speedFactor=Math.max(0,1-Math.abs(vt)/P.maxSpeed);
      Ft+=throttle*P.maxDrive*speedFactor;
    }
    if(w.drive && throttle<0){
      Ft+=throttle*P.maxDrive*0.45;
    }
    if(braking>0){
      const bs=Math.sign(vt);
      if(Math.abs(vt)>0.3) Ft-=bs*braking*P.maxBrake;
      else Ft-=vt*400;
    }
    // rolling resistance
    Ft-=vt*P.rollRes;

    const maxF=mu*Fn*1.35+40;
    Ft=clamp(Ft,-maxF,maxF);

    fx+=Ft*tx; fy+=Ft*ty;
    torque+=rx*(Ft*ty)-ry*(Ft*tx);

    bike.wheelSpin+= vt*dt/P.wheelR;
  }

  // air / lean control
  const onG=anyGround;
  if(!onG){
    torque+=lean*P.airTorque*P.mass*0.06;
    bike.airTime+=dt;
  } else {
    bike.airTime=0;
    torque+=lean*P.groundTorque*P.mass*0.035;
  }

  // angular damping
  torque-=bike.angVel*260;

  // integrate
  bike.vx+=fx/m*dt;
  bike.vy+=fy/m*dt;
  bike.angVel+=torque/I*dt;
  bike.angVel=clamp(bike.angVel,-14,14);
  bike.x+=bike.vx*dt;
  bike.y+=bike.vy*dt;
  bike.ang+=bike.angVel*dt;

  bike.onGround=onG;
  if(onG) bike.groundCount++; else bike.groundCount=0;
}

/* collision detection against hazards */
function checkCollisions(){
  if(bike.crashTimer>0) return;
  const cosA=Math.cos(bike.ang), sinA=Math.sin(bike.ang);
  const pts=[
    {x:bike.x-0.9*cosA+0.5*sinA, y:bike.y-0.9*sinA-0.5*cosA, r:0.42},
    {x:bike.x+0.9*cosA+0.5*sinA, y:bike.y+0.9*sinA-0.5*cosA, r:0.42},
    {x:bike.x, y:bike.y, r:0.5}
  ];
  for(const st of platformStates){
    const o=st.o;
    if(o.t==='mover'||o.t==='fan'||o.t==='boost') continue;
    if(o.t==='spring') continue;
    if(o.t==='saw'){
      const cx=o.x, cy=(o.ground? (groundAt(level,o.x)+ (o.r||1.2)) : (o.y||0)) + 0;
      const yr = o.ground ? (groundAt(level,o.x)+(o.r||1.2)) : cy;
      for(const p of pts){
        const d=Math.hypot(p.x-cx,p.y-yr);
        if(d < (o.r||1.2)*0.88+p.r) return crash();
      }
      continue;
    }
    if(o.t==='hammer'){
      const ax=o.x, ay=(o.ground?groundAt(level,o.x):0)+(o.anchorH||4.2);
      const ang=st.angle;
      const hx=ax+Math.sin(ang)*(o.len||3);
      const hy=ay-Math.cos(ang)*(o.len||3);
      for(const p of pts){
        if(Math.hypot(p.x-hx,p.y-hy) < 0.95+p.r) return crash();
      }
      continue;
    }
    if(o.t==='bar'){
      const bx=o.x, by=(groundAt(level,o.x))+(o.h||3)/2;
      const bw=0.35, bh=o.h||3;
      const ox=(o.off||0);
      for(const p of pts){
        if(Math.abs(p.x-(bx+ox*0))<bw/2+p.r && Math.abs(p.y-by)<bh/2+p.r) return crash();
      }
      continue;
    }
    // boxes / spikes
    let cx=o.x, cy, w,h;
    if(o.t==='spike'){ w=o.w||1.5; h=o.h||0.9; cy=(o.ground?groundAt(level,o.x):o.y||0)+h/2; }
    else { w=o.w||1; h=o.h||1; cy=(o.ground?groundAt(level,o.x):(o.y||0))+h/2; }
    for(const p of pts){
      const dx=Math.abs(p.x-cx)-(w/2);
      const dy=Math.abs(p.y-cy)-(h/2);
      if(dx<p.r && dy<p.r){
        const ddx=Math.max(dx,0), ddy=Math.max(dy,0);
        if(ddx*ddx+ddy*ddy < p.r*p.r) return crash();
      }
    }
  }
}

function crash(){
  if(bike.crashTimer>0) return;
  bike.crashTimer=0.001;
  Game.onCrash();
}

/* ══════════════════════════════════════════════════════════
   11. GAME
   ══════════════════════════════════════════════════════════ */
const Game={
  state:'menu',           // menu | playing | paused | result
  mode:null,
  level:null,
  bikeObj:null,
  rider:null,
  bikeParts:null,
  time:0,
  score:0,
  attempts:0,
  cpsReached:0,
  cps:[],
  lastCheckpoint:null,
  finished:false,
  crashTimer:0,
  shake:0,
  camPos:new THREE.Vector3(0,5,-14),
  camLook:new THREE.Vector3(0,1,0),
  ghostRec:[],ghostObj:null,ghostIdx:0,
  input:{gas:0,brake:0,lean:0,jump:false},
  hudTimer:0,
  fx:[],
  partPool:[],
  decoRng:null,
  bestGhost:null,

  /* ---------- START LEVEL ---------- */
  start(lv,opts={}){
    level=lv; Game.level=lv;
    Game.mode=lv.cat;
    clearWorld();
    SceneBuilder.build(lv);
    buildWorldState(lv);

    // bike
    const b=buildBike(); Game.bikeParts=b;
    Game.bikeObj=b.root;
    worldRoot.add(b.root);
    const r=buildRider(); Game.rider=r;
    b.root.add(r.root);

    // start pos
    const sx=lv.pts[0][0]+10;
    const sy=terrainY(lv.pts,sx)+2.4;
    resetBikeAt(sx,sy);
    Game.lastCheckpoint={x:sx,y:sy};
    Game.cps=(lv.cps||[]).map(x=>({x,hits:false}));
    Game.cpsReached=0;
    Game.time=0; Game.score=0; Game.finished=false; Game.crashTimer=0;
    Game.shake=0;
    Game.ghostRec=[]; Game.ghostIdx=0;
    if(Game.ghostObj){ worldRoot.remove(Game.ghostObj); Game.ghostObj=null; }
    if(lv.ghost && SaveManager.level(lv.id).best>0){
      const g=buildBike(); g.root.traverse(o=>{ if(o.isMesh){ o.material=o.material.clone(); o.material.transparent=true; o.material.opacity=0.3; o.material.depthWrite=false; } });
      Game.ghostObj=g.root; worldRoot.add(g.root);
      const rr=buildRider(); rr.root.traverse(o=>{ if(o.isMesh){ o.material=o.material.clone(); o.material.transparent=true; o.material.opacity=0.25; o.material.depthWrite=false; } });
      g.root.add(rr.root); Game.ghostRider=rr;
    }

    Game.camPos.set(sx-3,sy+3.5,11);
    Game.state='playing';
    UI.show('hud');
    UI.buildCpDots();
    AudioMan.init(); AudioMan.resume(); AudioMan.engineStart();
    AudioMan.startMusic(['factory','city','mountain','lab','ice','volcano','tunnel'].indexOf(lv.theme));

    SaveManager.addAttempt(lv.id);
    Game.tutorialStep=0;
  },

  /* ---------- RESTART ---------- */
  restart(){ Game.start(level,{}); },

  checkpointRespawn(){
    const c=Game.lastCheckpoint;
    resetBikeAt(c.x,c.y);
    bike.crashTimer=0;
    Game.crashTimer=0;
    Game.input.gas=0;Game.input.brake=0;Game.input.lean=0;
    Game.score=Math.max(0,Game.score-120);
  },

  onCrash(){
    if(Game.state!=='playing'||Game.finished) return;
    Game.crashTimer=1.15;
    bike.crashTimer=1;
    AudioMan.crash();
    Game.shake=1.0;
    vibrate([40,30,80]);
    spawnParticles(bike.x,bike.y,0xff8844,26,7);
    UI.toast('💥 '+t('failed'),'#ff6b6b');
  },

  finish(){
    if(Game.finished) return;
    Game.finished=true;
    AudioMan.win();
    vibrate([30,50,30,50,90]);
    const lv=Game.level;
    const targets=lv.targets||[90,70,55];
    let medal=0;
    if(Game.time<=targets[2]) medal=3;
    else if(Game.time<=targets[1]) medal=2;
    else if(Game.time<=targets[0]) medal=1;

    // score
    const timeScore=Math.max(0,Math.round(3000-Game.time*22));
    const cpScore=Game.cpsReached*250;
    const total=Math.max(0,timeScore+cpScore);
    Game.score+=total;

    const res=SaveManager.setResult(lv.id,Game.time,Game.score,medal);
    UI.showResult(true,medal,res);
  },

  /* ---------- UPDATE ---------- */
  update(dt){
    if(Game.state!=='playing') return;
    Game.time+=dt;

    if(Game.crashTimer>0){
      Game.crashTimer-=dt;
      if(Game.crashTimer<=0){
        if(Game.finished) return;
        Game.checkpointRespawn();
      }
      updateVisuals(dt*0.3,true);
      return;
    }

    // ---- physics substeps ----
    let acc=Math.min(dt,0.05);
    let steps=0;
    while(acc>0 && steps<8){
      const h=Math.min(FIXED,acc);
      stepPhysics(h);
      acc-=h; steps++;
    }

    // ---- hazards dynamics ----
    for(const st of platformStates){
      const o=st.o;
      st.time+=dt;
      if(o.t==='mover'){
        const t2=st.time*(o.spd||0.7);
        const off=Math.sin(t2)*((o.dy||3)/2);
        const prevY=st.y;
        st.y=(o.ground? ((groundAt(level,o.x)||0)+(o.h||0.7)/2) : st.baseY) + off;
        st.vy=(st.y-prevY)/Math.max(dt,0.0001);
      }
      if(o.t==='saw') st.angle=(st.angle||0)+(o.spd||2)*dt;
      if(o.t==='hammer') st.angle=(o.range||1)*Math.sin(st.time*(o.spd||1.5));
      if(o.t==='spring'){ if(st.cool>0) st.cool-=dt; st.squash=Math.max(0,st.squash-dt*4); }
    }

    // ---- fans ----
    for(const st of platformStates){
      const o=st.o;
      if(o.t!=='fan') continue;
      const cx=o.x, cy=(o.ground?groundAt(level,o.x):0)+(o.h||4)/2;
      const dx=bike.x-cx, dy=bike.y-cy;
      if(Math.abs(dx)<(o.w||4)/2+1 && Math.abs(dy)<(o.h||4)/2+1){
        bike.vx+=(o.force||0)*dt;
        if(Game.frameCount%3===0) spawnParticles(cx+ (o.force>0?1:-1),cy,0x99ddff,1,2);
      }
    }
    // ---- springs ----
    for(const st of platformStates){
      const o=st.o;
      if(o.t!=='spring'||st.cool>0) continue;
      const cy=groundAt(level,o.x)+0.6;
      if(Math.abs(bike.x-o.x)<1.1 && Math.abs(bike.y-cy)<1.4){
        bike.vy=Math.max(bike.vy,o.force||20);
        bike.angVel*=0.4;
        st.cool=0.5; st.squash=1;
        AudioMan.spring();
        vibrate(25);
      }
    }
    // ---- boosts ----
    for(const st of platformStates){
      const o=st.o;
      if(o.t!=='boost') continue;
      const cy=groundAt(level,o.x)+0.4;
      if(Math.abs(bike.x-o.x)<(o.w||2)/2 && Math.abs(bike.y-cy)<1.2){
        bike.vx=Math.max(bike.vx, 20);
        AudioMan.boost();
      }
    }

    // ---- collisions ----
    checkCollisions();
    if(Game.crashTimer>0){ updateVisuals(dt,true); return; }

    // ---- world bounds ----
    const minX=level.pts[0][0]+1;
    if(bike.x<minX){ bike.x=minX; bike.vx=Math.max(0,bike.vx); }

    const lowest=terrainY(level.pts,bike.x);
    if(bike.y< -40 || (isFinite(lowest)&&bike.y<lowest-25)){
      Game.onCrash();
      return;
    }

    // ---- checkpoints ----
    Game.cps.forEach((c,i)=>{
      if(!c.hits && bike.x>c.x){
        c.hits=true; Game.cpsReached++;
        const gy=terrainY(level.pts,c.x);
        Game.lastCheckpoint={x:c.x, y:gy+2.4};
        AudioMan.cp();
        vibrate(20);
        UI.toast('⚑ '+t('cpReached'),'#4ade80');
        UI.buildCpDots();
        Game.score+=250;
      }
    });

    // ---- finish ----
    if(!Game.finished && bike.x>=level.finish){
      Game.finish();
      return;
    }

    // ---- ghost record ----
    if(level.ghost){
      Game.ghostRec.push({x:bike.x,y:bike.y,a:bike.ang,t:Game.time});
    }

    // ---- HUD ----
    Game.hudTimer+=dt;
    if(Game.hudTimer>0.05){
      Game.hudTimer=0;
      UI.updateHUD();
    }

    updateVisuals(dt,false);
  }
};

/* ══════════════════════════════════════════════════════════
   12. VISUALS UPDATE

function updateVisuals(dt,crashed){
  const B=Game.bikeParts, R=Game.rider;
  if(!B||!R) return;

  // bike transform
  Game.bikeObj.position.set(bike.x,bike.y,0);
  Game.bikeObj.rotation.z=bike.ang;

  // wheel spin
  B.rearWheel.rotation.z=-bike.wheelSpin;
  B.frontWheel.rotation.z=-bike.wheelSpin*0.98;

  // crank
  const spd=Math.hypot(bike.vx,bike.vy);
  bike.crank+=spd*dt*1.1*(bike.vx>0.2?1:0);
  B.crankGroup.rotation.z=-bike.crank;

  // pedal positions (local to bike root)
  const cx=0.02, cy=-0.12, cr=0.35;
  const px=Math.cos(-bike.crank)*0+Math.sin(-bike.crank+Math.PI/2)*0;
  const pedalLx=cx+Math.cos(bike.crank+Math.PI/2)*0; // unused

  const pAngL=-bike.crank+Math.PI/2;
  const pAngR=-bike.crank-Math.PI/2;
  const pLx=cx+Math.cos(pAngL)*cr*0; // keep x small
  // pedals: crank arm rotates in XY plane
  const pL={x:cx+Math.sin(bike.crank)*cr, y:cy-Math.cos(bike.crank)*cr};
  const pR={x:cx-Math.sin(bike.crank)*cr, y:cy+Math.cos(bike.crank)*cr};

  // rider pose
  const airborne=!bike.onGround && bike.airTime>0.12;
  const leanInput=Game.input.lean;

  // pelvis local offset relative to bike
  const targetPelvisY=0.30 - (leanInput>0?0.03:0) ;
  R.pelvis.position.x=damp(R.pelvis.position.x,-0.28+leanInput*0.06,10,dt);
  R.pelvis.position.y=damp(R.pelvis.position.y,targetPelvisY,10,dt);
  R.torso.rotation.z=damp(R.torso.rotation.z, (airborne? -0.35: (0.32 + leanInput*0.22)) + (crashed?-0.5:0), 9, dt);
  R.head.rotation.z=damp(R.head.rotation.z, airborne? 0.25 : -0.22, 8, dt);

  // legs IK to pedals (local coords of pelvis)
  const pelvisLocal={x:R.pelvis.position.x, y:R.pelvis.position.y};
  const ikLeg=(limb,pedal,zoff)=>{
    // transform pedal into pelvis space: pelvis is not rotated, only translated
    const tx=pedal.x-pelvisLocal.x;
    const ty=pedal.y-pelvisLocal.y;
    solveIK(limb,tx,ty,0.46,0.46,false);
  };
  ikLeg(R.legL,pL,0.11);
  ikLeg(R.legR,pR,-0.11);

  // arms IK to handlebar (handlebar at 0.8,0.34 in bike space; torso space needs transform)
  const hbLocal={x:0.78,y:0.36};
  const ikArm=(limb)=>{
    // limb parent = torso which is child of pelvis at (0,0.12)
    const torsoWorldX=R.pelvis.position.x;
    const torsoWorldY=R.pelvis.position.y+0.12;
    // torso is rotated; approximate by ignoring rotation for IK target
    const tx=hbLocal.x-torsoWorldX-0.02;
    const ty=hbLocal.y-torsoWorldY-0.5;
    solveIK(limb,tx,ty,0.33,0.33,false);
  };
  ikArm(R.armL); ikArm(R.armR);

  if(crashed){
    Game.bikeObj.rotation.z=bike.ang+Math.sin(Game.time*30)*0.3;
  }

  // ---- camera ----
  const camSens=SaveManager.data.settings.camSens;
  const speedZoom=clamp(spd/26,0,1);
  const back=damp(6.2+speedZoom*2.2,0,0,0);
  const height=3.2+speedZoom*0.6;
  const side=10.5+speedZoom*1.8;
  const targetX=bike.x+2.2+bike.vx*0.10;
  const targetY=bike.y+1.0;
  const px2=damp(Game.camPos.x,targetX-3.2,4.5/camSens,dt);
  const py2=damp(Game.camPos.y,targetY+height,4.5/camSens,dt);
  const pz2=damp(Game.camPos.z,side,4.0/camSens,dt);
  Game.camPos.set(px2,py2,pz2);
  Game.camLook.set(damp(Game.camLook.x,targetX,6,dt), damp(Game.camLook.y,targetY,6,dt), 0);

  // shake
  Game.shake=Math.max(0,Game.shake-dt*2.4);
  const sh=Game.shake*0.5;
  camera.position.set(Game.camPos.x+(Math.random()-0.5)*sh, Game.camPos.y+(Math.random()-0.5)*sh, Game.camPos.z+(Math.random()-0.5)*sh*0.6);
  camera.lookAt(Game.camLook.x,Game.camLook.y,0);
  camera.position.z=Math.max(4, camera.position.z);

  // shadow follow
  sun.position.set(bike.x-18, bike.y+34, 18);
  sun.target.position.set(bike.x, bike.y, 0);
  sun.target.updateMatrixWorld();

  // ghost
  if(Game.ghostObj && level.ghost){
    const rec=Game.ghostRec;
    // replay previous best: we simply time-shift along recorded array
    const bt=SaveManager.level(level.id).best;
    if(bt>0){
      // simple: move ghost along current recorded path if it exists (real ghost needs stored data)
      const idx=Math.min(Game.ghostRec.length-1, Math.floor((Game.time/bt)*Game.ghostRec.length));
      if(idx>=0 && Game.ghostRec[idx]){
        const g=Game.ghostRec[idx];
        Game.ghostObj.position.set(g.x,g.y,0);
        Game.ghostObj.rotation.z=g.a;
      }
    }
  }

  // animated hazards visuals
  for(let i=0;i<platformStates.length;i++){
    const st=platformStates[i];
    const obj=st.mesh;
    if(!obj) continue;
    const o=st.o;
    if(o.t==='saw'&&obj.userData.spin){ obj.userData.spin.rotation.z=st.angle; }
    if(o.t==='hammer'&&obj.userData.anchor){ obj.userData.anchor.rotation.z=st.angle; }
    if(o.t==='mover'){ obj.position.y=st.y; }
    if(o.t==='fan'&&obj.userData.blade){ obj.userData.blade.rotation.x+=dt*14; }
    if(o.t==='spring'&&obj.userData.springCoil){ obj.userData.springCoil.scale.y=1-st.squash*0.6; }
  }

  // particles
  updateParticles(dt);
}

const particlePool=[];
const particleGeo=new THREE.SphereGeometry(0.09,5,4);
function spawnParticles(x,y,color,count,spread){
  if(quality==='low') count=Math.ceil(count*0.4);
  for(let i=0;i<count;i++){
    let p=particlePool.find(q=>!q.alive);
    if(!p){
      if(particlePool.length>140) break;
      const m=new THREE.Mesh(particleGeo,new THREE.MeshBasicMaterial({color:color,transparent:true}));
      m.userData.shared=false;
      worldRoot.add(m);
      p={mesh:m,alive:false}; particlePool.push(p);
    }
    p.alive=true; p.life=0.55+Math.random()*0.4; p.maxLife=p.life;
    p.mesh.visible=true;
    p.mesh.position.set(x+(Math.random()-0.5)*0.5,y+(Math.random()-0.5)*0.5,(Math.random()-0.5)*0.8);
    p.vx=(Math.random()-0.5)*spread;
    p.vy=(Math.random()*0.7+0.3)*spread*0.6;
    p.mesh.scale.setScalar(0.6+Math.random()*0.9);
  }
}
function updateParticles(dt){
  for(const p of particlePool){
    if(!p.alive) continue;
    p.life-=dt;
    if(p.life<=0){ p.alive=false; p.mesh.visible=false; continue; }
    p.vy-=18*dt;
    p.mesh.position.x+=p.vx*dt;
    p.mesh.position.y+=p.vy*dt;
    const a=p.life/p.maxLife;
    p.mesh.material.opacity=a;
    p.mesh.scale.setScalar(0.3+a*0.9);
  }
}

        else if(o.t==='box') py=gg+(o.h||1)/2;

    }

    // finish gate
    const fx=lv.finish;
    const fy=terrainY(lv.pts,fx);
    const gate=new THREE.Group();
    const postMat=mat(0xdddddd,{metalness:0.7,roughness:0.3});
    const p1=new THREE.Mesh(new THREE.CylinderGeometry(0.16,0.16,5,8),postMat); p1.position.set(0,2.5,-4);
    const p2=new THREE.Mesh(new THREE.CylinderGeometry(0.16,0.16,5,8),postMat); p2.position.set(0,2.5,4);
    const banner=new THREE.Mesh(new THREE.BoxGeometry(0.3,0.9,8.4),new THREE.MeshStandardMaterial({color:0xffffff,emissive:0x666666,emissiveIntensity:0.5}));
    banner.position.y=4.4;
    gate.add(p1,p2,banner);
    gate.position.set(fx,fy,0);
    worldRoot.add(gate);

    // checkpoints
    (lv.cps||[]).forEach((cx,i)=>{
      const cy=terrainY(lv.pts,cx);
      const cpg=new THREE.Group();
      const mat1=new THREE.MeshStandardMaterial({color:0x22dd88,emissive:0x116644,emissiveIntensity:0.8,transparent:true,opacity:0.8});
      const arch=new THREE.Mesh(new THREE.TorusGeometry(2.6,0.13,8,20,Math.PI),mat1);
      arch.position.y=0.2;
      cpg.add(arch);
      cpg.position.set(cx,cy,0);
      cpg.rotation.y=Math.PI/2;
      worldRoot.add(cpg);
    });
  }
};

function bindTouch(){
  const bind=(id,on,off)=>{
    const el=$(id); if(!el) return;
    const start=e=>{ e.preventDefault(); el.classList.add('press'); on(); };
    const end=e=>{ e.preventDefault(); el.classList.remove('press'); off(); };

  };

function doJump(){
  if(Game.state!=='playing') return;
  if(bike.onGround && bike.groundCount>1){
    bike.vy+=7.2;
    bike.angVel+=Game.input.lean*1.6;
    AudioMan.jump();
    vibrate(18);
    spawnParticles(bike.x,bike.y-0.6,0xcccccc,8,3);
  } else {
    bike.angVel+=Game.input.lean*0.8;
  }
}

      this.groundPts.sort((a,b)=>a[0]-b[0]);
      this.rebuildGround();

    worldRoot.add(g);

let editorListeners=[];
function setupEditorListeners(){
  const onMove=e=>{
    const cx=(e.touches?e.touches[0].clientX:e.clientX);
    const cy=(e.touches?e.touches[0].clientY:e.clientY);
    Editor.mouse.x=(cx/innerWidth)*2-1;
    Editor.mouse.y=-(cy/innerHeight)*2+1;
  };
  const onClick=e=>{
    if(!Editor.active) return;
    const cx=(e.changedTouches?e.changedTouches[0].clientX:e.clientX);
    const cy=(e.changedTouches?e.changedTouches[0].clientY:e.clientY);
    // raycast against ground plane z=0
    const ndc=new THREE.Vector2((cx/innerWidth)*2-1, -(cy/innerHeight)*2+1);
    Editor.ray.setFromCamera(ndc,camera);
    const plane=new THREE.Plane(new THREE.Vector3(0,0,1),0);
    const pt=new THREE.Vector3();
    if(Editor.ray.ray.intersectPlane(plane,pt)){
      // check if hit existing object visual
      const hits=Editor.ray.intersectObjects(Editor.visuals,true);
      if(hits.length){
        let obj=hits[0].object;
        while(obj.parent && !Editor.visuals.includes(obj)) obj=obj.parent;
        const idx=Editor.visuals.indexOf(obj);
        if(idx>=0){ Editor.selected=idx; Editor.renderObjects(); return; }
      }
      Editor.addObject(pt.x,pt.y);
    }
  };
  const onKey=e=>{
    if(!Editor.active) return;
    if(e.key==='Delete'||e.key==='Backspace'){ Editor.deleteSelected(); }
    if(e.key==='z'&&(e.ctrlKey||e.metaKey)&&!e.shiftKey){ e.preventDefault(); Editor.undo(); }
    if((e.key==='y'&&(e.ctrlKey||e.metaKey))||(e.key==='z'&&e.ctrlKey&&e.shiftKey)){ e.preventDefault(); Editor.redo(); }
    if(e.key==='r'&&Editor.selected>=0){ Editor.objects[Editor.selected].rot=(Editor.objects[Editor.selected].rot||0)+0.3; Editor.renderObjects(); }
    // move selected with arrows
    if(Editor.selected>=0){
      const o=Editor.objects[Editor.selected];
      const step=e.shiftKey?0.5:0.25;
      if(e.key==='ArrowRight'){ o.x+=step; Editor.renderObjects(); }
      if(e.key==='ArrowLeft'){ o.x-=step; Editor.renderObjects(); }
      if(e.key==='ArrowUp'){ o.y=(o.y||0)+step; Editor.renderObjects(); }
      if(e.key==='ArrowDown'){ o.y=(o.y||0)-step; Editor.renderObjects(); }
    }
  };
  const onWheel=e=>{
    if(!Editor.active) return;
    e.preventDefault();
    const z=clamp(camera.position.z+e.deltaY*0.05,15,140);
    camera.position.z=z;
    camera.updateProjectionMatrix();
  };
  canvas.addEventListener('pointermove',onMove);
  canvas.addEventListener('pointerdown',onClick);
  addEventListener('keydown',onKey);
  canvas.addEventListener('wheel',onWheel,{passive:false});
  editorListeners=[['pointermove',onMove],['pointerdown',onClick],['keydown',onKey],['wheel',onWheel]];
}
function removeEditorListeners(){
  canvas.removeEventListener('pointermove',editorListeners[0][1]);
  canvas.removeEventListener('pointerdown',editorListeners[1][1]);
  removeEventListener('keydown',editorListeners[2][1]);
  canvas.removeEventListener('wheel',editorListeners[3][1]);
}

/* ══════════════════════════════════════════════════════════
   19. MAIN LOOP
   ══════════════════════════════════════════════════════════ */
let lastT=performance.now();
let frames=0,fpsAcc=0,fps=60;
function loop(now){
  requestAnimationFrame(loop);
  const dt=Math.min(0.05,(now-lastT)/1000);
  lastT=now;
  Game.frameCount=(Game.frameCount||0)+1;

  // input
  if(Game.state==='playing'){
    if(isTouch){
      Game.input.gas=touchState.gas;
      Game.input.brake=touchState.brake;
      Game.input.lean=touchState.lean;
    } else {
      const k=readKeyboard();
      Game.input.gas=k.gas; Game.input.brake=k.brake; Game.input.lean=k.lean;
    }
    Game.update(dt);
    // engine audio
    AudioMan.engineUpdate(Math.hypot(bike.vx,bike.vy),Game.input.gas,bike.onGround);
    // auto low quality on bad fps
    frames++; fpsAcc+=dt;
    if(fpsAcc>1){ fps=frames/fpsAcc; frames=0; fpsAcc=0;
      if(fps<32 && quality==='high'){ quality='medium'; applyQuality(); SaveManager.data.settings.quality='medium'; SaveManager.save(); }
    }
  } else if(Editor.active){
    // slow orbit
    const t2=now*0.00008;
    camera.position.x=Editor.groundPts[0][0]+ (Editor.groundPts[Editor.groundPts.length-1][0]-Editor.groundPts[0][0])*0.5 + Math.sin(t2)*10;
    camera.lookAt(camera.position.x, 4, 0);
  }

  renderer.render(scene,camera);
}

    case 'resume': UI.resume(); break;
    case 'restart': UI.resume(); Game.restart(); break;
    case 'quit': UI.quit(); break;
    // editor
    case 'editorExit': Editor.exit(); break;
    case 'editorPlay': Editor.playtest(); break;
    case 'editorUndo': Editor.undo(); break;
    case 'editorRedo': Editor.redo(); break;
    case 'editorDelete': Editor.deleteSelected(); break;
    case 'editorSave': Editor.save(); break;
    case 'editorClear': Editor.clearAll(); break;
  }
});

/* settings binding */
function initSettings(){
  const s=SaveManager.data.settings;
  $('#sMusic').value=s.music; $('#sSfx').value=s.sfx;
  $('#sVib').classList.toggle('on',s.vibration);
  $('#sQual').value=s.quality;
  $('#sCam').value=s.camSens; $('#sCtl').value=s.ctlSens;
  $('#sBtn').value=s.btnOpacity;
  $('#sLang').value=s.lang;
  $('#sMusic').oninput=e=>{ s.music=+e.target.value; AudioMan.setMusicVol(s.music); SaveManager.save(); };
  $('#sSfx').oninput=e=>{ s.sfx=+e.target.value; AudioMan.setSfxVol(s.sfx); SaveManager.save(); };
  $('#sVib').onclick=()=>{ s.vibration=!s.vibration; $('#sVib').classList.toggle('on',s.vibration); SaveManager.save(); };
  $('#sQual').onchange=e=>{ s.quality=e.target.value; quality=s.quality; applyQuality(); SaveManager.save(); };
  $('#sCam').oninput=e=>{ s.camSens=+e.target.value; SaveManager.save(); };
  $('#sCtl').oninput=e=>{ s.ctlSens=+e.target.value; SaveManager.save(); };
  $('#sBtn').oninput=e=>{ s.btnOpacity=+e.target.value; applyBtnOpacity(); SaveManager.save(); };
  $('#sLang').onchange=e=>{ s.lang=e.target.value; applyI18n(); UI.renderTabs(); SaveManager.save(); };
}
function applyBtnOpacity(){
  const v=SaveManager.data.settings.btnOpacity;
  $$('.tbtn').forEach(b=>{
    const base=b.id==='tGas'?'rgba(40,160,90,':b.id==='tBrake'?'rgba(200,60,60,':b.id==='tJump'?'rgba(60,120,240,':'rgba(120,90,220,';
    b.style.background=base+(0.15+v*0.5)+')';
  });
}

const LOAD_STEPS=[
  'در حال آماده‌سازی موتور گرافیکی…',
  'ساخت مدل دوچرخه…',
  'طراحی شخصیت…',
  'ساخت محیط‌ها…',
  'چیدمان مراحل…',
  'آماده!'
];
let loadStep=0;
function boot(){
  const fill=$('#barfill'), txt=$('#loadtxt');
  const step=()=>{
    loadStep++;
    const p=Math.min(100,loadStep/LOAD_STEPS.length*100);
    fill.style.width=p+'%';
    txt.textContent=LOAD_STEPS[Math.min(loadStep,LOAD_STEPS.length-1)];
    if(loadStep<LOAD_STEPS.length){
      setTimeout(step,120+Math.random()*160);
    } else {
      setTimeout(()=>{
        $('#loading').classList.remove('on');
        UI.show('menu');
        applyI18n();
        initSettings();
        applyBtnOpacity();
        bindTouch();
        updateRotateHint();
      },320);
    }
  };
  setTimeout(step,300);
}

function updateRotateHint(){
  const isPortrait = innerHeight>innerWidth && innerWidth<820;
  $('#rotate').classList.toggle('on', isPortrait && isTouch);
}
addEventListener('resize',updateRotateHint);
addEventListener('orientationchange',()=>setTimeout(updateRotateHint,300));
